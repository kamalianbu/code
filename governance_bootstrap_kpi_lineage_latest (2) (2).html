def qualified(table: str) -> str:
    """Return fully qualified Delta table name: Lakehouse.Schema.Table."""
    return f"{GOV_LAKEHOUSE}.{GOV_SCHEMA}.{table}"
def ensure_schema():
    """Create the governance schema if it does not already exist."""
    spark.sql(
        f"CREATE SCHEMA IF NOT EXISTS "
        f"{GOV_LAKEHOUSE}.{GOV_SCHEMA}"
    )
def create_delta_if_absent(table: str,ddl_columns: str,partition: str = ""):
    """
    Create a Delta table if it does not already exist.
    Automatically adds RunId and IngestedAtUtc audit columns.
    """
    part_clause = (
        f" PARTITIONED BY ({partition})"
        if partition
        else ""
    )
    spark.sql(f"""
        CREATE TABLE IF NOT EXISTS {qualified(table)} (
            {ddl_columns},
            RunId         STRING,
            IngestedAtUtc TIMESTAMP
        )
        USING DELTA
        {part_clause}
    """)
def upsert(
    pdf: pd.DataFrame,
    table: str,
    keys: list
):
    """
    Merge a pandas DataFrame into a Delta table.

    Adds RunId and IngestedAtUtc.
    Updates and inserts only columns available in both
    the source DataFrame and target Delta table.
    """
    if pdf is None or pdf.empty:
        print(f"  ↷ {table}: no rows to write")
        return
    pdf = pdf.copy()
    pdf["RunId"] = RUN_ID
    pdf["IngestedAtUtc"] = RUN_TS
    sdf = spark.createDataFrame(pdf)
    tgt = DeltaTable.forName(
        spark,
        qualified(table)
    )
    # Identify columns available in the target table.
    target_columns = set(
        spark.table(
            qualified(table)
        ).columns
    )
    # Use only columns available in both source and target.
    common_columns = [
        col_name
        for col_name in sdf.columns
        if col_name in target_columns
    ]
    # Confirm all merge-key columns are available.
    missing_keys = [
        key
        for key in keys
        if key not in common_columns
    ]
    if missing_keys:
        raise ValueError(
            f"{table}: merge key column(s) missing "
            f"from source or target: {missing_keys}"
        )
    # Update non-key columns only.
    update_values = {
        col_name: f"s.`{col_name}`"
        for col_name in common_columns
        if col_name not in keys
    }
    # Insert all common source and target columns.
    insert_values = {
        col_name: f"s.`{col_name}`"
        for col_name in common_columns
    }
    join_cond = " AND ".join(
        f"t.`{key}` = s.`{key}`"
        for key in keys
    )
    merge_cmd = (
        tgt.alias("t")
        .merge(
            sdf.alias("s"),
            join_cond
        )
    )
    if update_values:
        merge_cmd = merge_cmd.whenMatchedUpdate(
            set=update_values
        )
    merge_cmd = merge_cmd.whenNotMatchedInsert(
        values=insert_values
    )
    merge_cmd.execute()
    print(
        f"  ✔ {table}: "
        f"{len(pdf)} rows merged"
    )
def resolve_workspace_id(ws_name_or_id: str) -> str:
    """
    Resolve a workspace name or GUID to its GUID.
    Raises ValueError when the workspace is not accessible.
    """
    df = fabric.list_workspaces()
    search_value = str(
        ws_name_or_id
    ).lower()
    mask = (
        df["Name"].str.lower() == search_value
    ) | (
        df["Id"]
        .astype(str)
        .str.lower() == search_value
    )
    if df[mask].empty:
        raise ValueError(f"Workspace '{ws_name_or_id}' "f"is not accessible.")
    return str(df[mask]["Id"].iloc[0])

bridge_table = qualified("bridge_domain_kpi")

bridge_df = spark.table(bridge_table)


# ---------------------------------------------------------
# 2. Create a backup before deleting anything
# ---------------------------------------------------------

backup_suffix = datetime.utcnow().strftime(
    "%Y%m%d_%H%M%S"
)

backup_table = qualified(
    f"bridge_domain_kpi_backup_{backup_suffix}"
)

(
    bridge_df
    .write
    .format("delta")
    .mode("overwrite")
    .saveAsTable(backup_table)
)

print(
    f"Backup created successfully: {backup_table}"
)


# ---------------------------------------------------------
# 3. Find KPIs currently mapped to multiple domains
# ---------------------------------------------------------

conflicting_kpis = (
    bridge_df
    .groupBy(
        "DatasetId",
        "KPIName"
    )
    .agg(
        F.countDistinct("DomainCode").alias(
            "DomainCount"
        ),
        F.collect_set("DomainCode").alias(
            "Domains"
        )
    )
    .filter(
        F.col("DomainCount") > 1
    )
)

conflict_count = conflicting_kpis.count()

print(
    f"KPIs with multiple domain mappings: "
    f"{conflict_count}"
)

display(conflicting_kpis)


# ---------------------------------------------------------
# 4. Rank only the conflicting mappings
#
# The latest IngestedAtUtc row is treated as authoritative.
# MappedOn is used as the second ordering field.
# ---------------------------------------------------------

conflicting_rows = (
    bridge_df.alias("bridge")
    .join(
        conflicting_kpis
        .select(
            "DatasetId",
            "KPIName"
        )
        .alias("conflicts"),
        on=[
            "DatasetId",
            "KPIName"
        ],
        how="inner"
    )
)

latest_mapping_window = (
    Window
    .partitionBy(
        "DatasetId",
        "KPIName"
    )
    .orderBy(
        F.col("IngestedAtUtc").desc_nulls_last(),
        F.col("MappedOn").desc_nulls_last(),
        F.col("DomainCode").asc()
    )
)

ranked_conflicting_rows = (
    conflicting_rows
    .withColumn(
        "MappingRank",
        F.row_number().over(
            latest_mapping_window
        )
    )
)


# Latest mappings that will be retained
mappings_to_keep = (
    ranked_conflicting_rows
    .filter(
        F.col("MappingRank") == 1
    )
)

print("Latest mappings that will be retained:")

display(
    mappings_to_keep.select(
        "DatasetId",
        "KPIName",
        "DomainCode",
        "MappingSource",
        "MappedOn",
        "IngestedAtUtc"
    )
)


# Older conflicting mappings that will be deleted
mappings_to_delete = (
    ranked_conflicting_rows
    .filter(
        F.col("MappingRank") > 1
    )
    .select(
        "DomainCode",
        "DatasetId",
        "KPIName"
    )
    .dropDuplicates()
)

delete_count = mappings_to_delete.count()

print(
    f"Older mappings identified for deletion: "
    f"{delete_count}"
)

display(mappings_to_delete)


# ---------------------------------------------------------
# 5. Delete only the older conflicting mappings
# ---------------------------------------------------------

if delete_count > 0:

    bridge_delta = DeltaTable.forName(
        spark,
        bridge_table
    )

    (
        bridge_delta.alias("target")
        .merge(
            mappings_to_delete.alias("old"),
            """
            target.DomainCode = old.DomainCode
            AND target.DatasetId = old.DatasetId
            AND target.KPIName = old.KPIName
            """
        )
        .whenMatchedDelete()
        .execute()
    )

    print(
        f"Deleted {delete_count} outdated "
        f"KPI-domain mappings."
    )

else:
    print(
        "No outdated KPI-domain mappings "
        "were found."
    )


# ---------------------------------------------------------
# 6. Validate the bridge after cleanup
# ---------------------------------------------------------

remaining_conflicts = (
    spark.table(bridge_table)
    .groupBy(
        "DatasetId",
        "KPIName"
    )
    .agg(
        F.countDistinct("DomainCode").alias(
            "DomainCount"
        ),
        F.collect_set("DomainCode").alias(
            "Domains"
        )
    )
    .filter(
        F.col("DomainCount") > 1
    )
)

remaining_conflict_count = (
    remaining_conflicts.count()
)

print(
    f"Remaining multi-domain KPI conflicts: "
    f"{remaining_conflict_count}"
)

display(remaining_conflicts)

Lineage Discovery¶

This is the main discovery step of the notebook. It starts with the KPI Dashboard and identifies all the connected reports and semantic models behind it. The notebook can discover lineage from a report, folder, or dashboard depending on how the KPI Dashboard is structured.

Once the semantic models are found, it checks the workspaces that the current user can access and determines where each semantic model is hosted. If a semantic model exists in a workspace that is not accessible, it is marked as not found.

Finally, the relationships between the KPI Dashboard, reports, and semantic models are stored in the bridge_report_dataset table, which is used later for metadata extraction, governance reporting, and lineage analysis.

def resolve_domain_for_dataset(
    dataset_id: str,
    dataset_name: str
):
    """
    Resolve the domain for one semantic model.

    Priority:
    1. Manual mapping by DatasetName: confidence 1.00
    2. Manual mapping by DatasetId: confidence 1.00
    3. Dataset-name pattern: confidence 0.70
    4. No match: Unmapped with confidence 0.00
    """

    dataset_id_key = str(dataset_id or "").strip().lower()
    dataset_name_key = str(dataset_name or "").strip().lower()

    # 1. Manual mapping by semantic-model name
    if dataset_name_key in manual_domain_map:
        return (
            manual_domain_map[dataset_name_key],
            "Manual",
            1.00,
        )

    # 2. Manual mapping by semantic-model ID
    if dataset_id_key in manual_domain_map:
        return (
            manual_domain_map[dataset_id_key],
            "Manual",
            1.00,
        )

    # 3. Controlled semantic-model-name matching
    for domain_code, patterns in DATASET_NAME_DOMAIN_PATTERNS:
        if any(
            str(pattern).lower() in dataset_name_key
            for pattern in patterns
        ):
            return (
                domain_code,

KPI to Domain Mapping¶

This cell assigns each KPI to a business domain. It first checks the KPI's Display Folder for a matching domain. If no match is found, it inherits the domain that was already assigned to the semantic model containing that KPI.

Any KPIs that still cannot be classified are marked as unmapped for review. The final KPI-to-domain mappings are then stored in the bridge_domain_kpi table and are used later for governance reporting, KPI analysis, and domain-level reporting.

            matched_source = "DatasetDomain"

from pyspark.sql import functions as F
from delta.tables import DeltaTable

GOV = "DataGovernance_metadata_lh.governance"

def add_concat_column(fq_table: str, new_col: str, parts: list[str], sep: str = "|"):
    """Idempotently add/refresh a concat surrogate column on a Delta table."""
    df = spark.read.table(fq_table)
    if new_col not in df.columns:
        # Widen the schema first (writes an ALTER TABLE ADD COLUMN under the hood)
        spark.sql(f"ALTER TABLE {fq_table} ADD COLUMNS ({new_col} STRING)")

    # Rebuild the surrogate value on every run so late-arriving rows are covered
    expr = F.concat_ws(sep, *[F.col(p).cast("string") for p in parts])
    (spark.read.table(fq_table)
        .withColumn(new_col, expr)
        .write.format("delta").mode("overwrite")
        .option("overwriteSchema","true")
        .saveAsTable(fq_table))
    print(f"✓ {fq_table}.{new_col} materialised")

def add_date_key_if_absent(
    table_name: str,
    source_column: str,
    new_column: str
):
    """
    Add an integer date-key column to an existing Delta table
    and populate it from a date or timestamp column.

    The generated format is yyyyMMdd.

    Example:
    2026-09-30 13:19:42 becomes 20260930.

    Safe to rerun.
    """

    full_table_name = qualified(table_name)

    existing_columns = {
        field.name
        for field in spark.table(
            full_table_name
        ).schema.fields
    }

    # Add the integer date-key column only when it is missing
    if new_column not in existing_columns:
        spark.sql(f"""
            ALTER TABLE {full_table_name}
            ADD COLUMNS (`{new_column}` INTEGER)
        """)

        print(
            f"Added {new_column} to {full_table_name}"
        )
    else:
        print(
            f"{new_column} already exists in "
            f"{full_table_name}"
        )

    # Populate or refresh the date-key value
    spark.sql(f"""
        UPDATE {full_table_name}

        SET `{new_column}` = CAST(
            DATE_FORMAT(
                `{source_column}`,
                'yyyyMMdd'
            ) AS INTEGER
        )

        WHERE `{source_column}` IS NOT NULL
    """)

    print(f"{full_table_name}.{new_column} populated " f"from {source_column}")
